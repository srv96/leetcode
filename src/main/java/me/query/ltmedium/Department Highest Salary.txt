select a.department as Department, a.name as Employee, a.salary as Salary
from (select d.name department, e.name name, e.salary salary
      from DEPARTMENT d
               inner join employee e on d.id = e.departmentId) a
         inner join (select join_t2.department, max(join_t2.salary) as salary
                     from (select d.name department, e.salary salary
                           from DEPARTMENT d
                                    inner join employee e on d.id = e.departmentId) join_t2
                     group by join_t2.department) b on a.department = b.department and a.salary=b.salary;

--OPTIMISED

WITH DepartmentEmployee AS (
    SELECT d.name AS department, e.name AS employee, e.salary
    FROM DEPARTMENT d
             INNER JOIN employee e ON d.id = e.departmentId
)
select a.department as Department, a.employee as Employee, a.salary as Salary
from (DepartmentEmployee) a inner join (select join_t2.department, max(join_t2.salary) as salary
                     from (DepartmentEmployee) join_t2
                     group by join_t2.department) b on a.department = b.department and a.salary=b.salary;
