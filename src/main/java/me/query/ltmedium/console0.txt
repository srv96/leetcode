CREATE FUNCTION getNthHighestSalary(N IN NUMBER) RETURN NUMBER IS
    result NUMBER;
BEGIN
    SELECT salary
    INTO result
    FROM (
             SELECT salary, DENSE_RANK() OVER (ORDER BY salary DESC) AS rank
             FROM Employee
         ) rank_table
    WHERE rank_table.rank = N;

    RETURN result;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RETURN NULL; -- Return NULL if no salary is found for the given rank
END;
