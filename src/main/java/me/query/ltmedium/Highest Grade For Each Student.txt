select *
from Enrollments;
select student_id, course_id, grade, RANK() over (PARTITION BY student_id order by grade desc) s_rnk from Enrollments;

with student_course_rank AS (select student_id,
                                    course_id,
                                    grade,
                                    RANK() over (PARTITION BY student_id order by grade desc, course_id) s_rnk
                             from Enrollments)
select student_id, course_id, grade
from student_course_rank
where s_rnk = 1;
