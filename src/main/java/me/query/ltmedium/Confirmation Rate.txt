With SignupsConfirmationsLeftJoin as (

)

With SignupsConfirmationsLeftJoin as (
    select s.user_id user_id, c.time_stamp time_stamp,c.action action from Signups s left join Confirmations c on s.user_id = c.user_id
)
select user_id , count(time_stamp) count from SignupsConfirmationsLeftJoin group by (user_id);

With SignupsConfirmationsLeftJoin as (select s.user_id user_id, c.time_stamp time_stamp, c.action action
                                      from Signups s
                                               left join Confirmations c on s.user_id = c.user_id),
     totalGroup as (select user_id, count(time_stamp) count from SignupsConfirmationsLeftJoin group by (user_id)),
     individualGroup as (select user_id, action, count(time_stamp) count
                         from SignupsConfirmationsLeftJoin
                         group by (user_id, action)),
     indvtotjoin as (select i.user_id user_id, i.action action, i.count indvCount, t.count totCount
                     from individualGroup i
                              inner join totalGroup t on t.user_id = i.user_id),
     itrank as (select it.user_id                                         user_id,
                       it.action                                          action,
                       it.indvCount                                       indvCount,
                       it.totCount                                        totCount,
                       Rank() over (partition by user_id order by action) rank
                from indvtotjoin it),
     alresult as (SELECT user_id,
                         action,
                         rank,
                         CASE
                             WHEN action = 'confirmed' THEN
                                 CASE
                                     WHEN totCount > 0 THEN ROUND((indvCount * 1.0 / totCount), 2)
                                     ELSE 0
                                     END
                             ELSE
                                 CASE
                                     WHEN totCount > 0 THEN ROUND(((totCount - indvCount) * 1.0 / totCount), 2)
                                     ELSE 0
                                     END
                             END AS confirmation_rate
                  FROM itrank)
select user_id, confirmation_rate
from alresult
where rank < 2;
