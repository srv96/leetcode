with empmgrtbl as (select b.id id, b.name name, b.department department, a.id managerId
                   from employees a
                            inner join employees b on a.id = b.managerId)
        ,
     directCount as (select managerId, count(id) as directs
                     from (select b.id id, b.name name, b.department department, a.id managerId
                           from employees a
                                    inner join employees b on a.id = b.managerId) empmgrtbl
                     group by managerId)


select name
from employees
         inner join (select managerId, count(id) as directs
                     from (select b.id id, b.name name, b.department department, a.id managerId
                           from employees a
                                    inner join employees b on a.id = b.managerId) empmgrtbl
                     group by managerId) directCount on employees.id = directCount.managerId
where directCount.directs >= 5;
